import asyncio, websockets, json from pymavlink import mavutil master = mavutil.mavlink_connection('/dev/ttyACM0', baud=115200) print("Waiting for heartbeat...") master.wait_heartbeat() print("✅ Connected") def set_mode(mode): master.set_mode(master.mode_mapping()[mode]) def arm(): set_mode("GUIDED") master.arducopter_arm() master.motors_armed_wait() print("✅ Armed") def disarm(): master.arducopter_disarm() def wait_for_gps(): print("Waiting GPS...") while True: msg = master.recv_match(type='GPS_RAW_INT', blocking=True) if msg.fix_type >= 3: print("✅ GPS LOCK") break def send_rc(t,y,p,r): def m(v): return int(1500+v*500) master.mav.rc_channels_override_send( master.target_system, master.target_component, m(r), m(p), m(t), m(y), 0,0,0,0 ) async def telemetry(ws): while True: msg = master.recv_match(blocking=False) if msg: if msg.get_type()=="ATTITUDE": await ws.send(json.dumps({"type":"telemetry","yaw":msg.yaw})) if msg.get_type()=="GLOBAL_POSITION_INT": await ws.send(json.dumps({ "type":"telemetry", "alt":msg.relative_alt/1000, "heading":msg.hdg/100 })) await asyncio.sleep(0.05) async def status(ws): while True: await ws.send(json.dumps({ "type":"status", "state":{ "armed": master.motors_armed(), "flying": False }, "mode": master.flightmode })) await asyncio.sleep(1) async def connect(): uri="wss://jdk-server.onrender.com/control" while True: try: async with websockets.connect(uri) as ws: print("🌐 Connected") asyncio.create_task(telemetry(ws)) asyncio.create_task(status(ws)) while True: msg = json.loads(await ws.recv()) if msg["type"]=="action": a = msg["action"] if a=="arm": arm() elif a=="disarm": disarm() elif a=="takeoff": alt = msg.get("alt",2) wait_for_gps() arm() print(f"Takeoff to {alt}m") master.mav.command_long_send( master.target_system, master.target_component, mavutil.mavlink.MAV_CMD_NAV_TAKEOFF, 0, 0,0,0,0,0,0, alt ) # wait until altitude reached while True: msg_alt = master.recv_match(type='GLOBAL_POSITION_INT', blocking=True) current_alt = msg_alt.relative_alt / 1000.0 if current_alt >= alt * 0.9: break print("Switching to LOITER") set_mode("LOITER") elif a=="land": set_mode("LAND") elif msg["type"]=="control": send_rc( msg["throttle"], msg["yaw"], msg["pitch"], msg["roll"] ) except Exception as e: print("Error:", e) await asyncio.sleep(5) asyncio.run(connect())
